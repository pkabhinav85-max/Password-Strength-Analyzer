import hashlib
import re
import secrets
import sqlite3
import string

DB_NAME = "password_history.db"


def init_db():
    """Initializes a local SQLite database to store password hashes securely."""
    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()
    # Store user_id and the hashed password to prevent plain text exposure
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS password_history (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            user_id TEXT NOT NULL,
            password_hash TEXT NOT NULL
        )
    ''')
    conn.commit()
    conn.close()


def hash_password(password):
    """Generates a secure SHA-256 hash of the password."""
    return hashlib.sha256(password.encode()).hexdigest()


def is_password_reused(user_id, password):
    """Checks if the password hash already exists in the user's history."""
    current_hash = hash_password(password)

    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()
    cursor.execute(
        "SELECT 1 FROM password_history WHERE user_id = ? AND password_hash = ?",
        (user_id, current_hash)
    )
    result = cursor.fetchone()
    conn.close()

    return result is not None


def save_password_hash(user_id, password):
    """Saves the new password hash into the database history."""
    current_hash = hash_password(password)

    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()
    cursor.execute(
        "INSERT INTO password_history (user_id, password_hash) VALUES (?, ?)",
        (user_id, current_hash)
    )
    conn.commit()
    conn.close()


def analyze_password(password):
    score = 0
    feedback = []

    # 1. Length Check
    if len(password) >= 12:
        score += 2
    elif len(password) >= 8:
        score += 1
    else:
        feedback.append(
            "• Password is too short (Minimum 8 characters, 12+ recommended).")

    # 2. Complexity Checks
    if re.search(r"[A-Z]", password):
        score += 1
    else:
        feedback.append("• Missing uppercase letters (A-Z).")

    if re.search(r"[a-z]", password):
        score += 1
    else:
        feedback.append("• Missing lowercase letters (a-z).")

    if re.search(r"\d", password):
        score += 1
    else:
        feedback.append("• Missing numbers (0-9).")

    if re.search(r"[!@#$%^&*(),.?\":{}|<>_+-]", password):
        score += 1
    else:
        feedback.append("• Missing special characters (e.g., !, @, #).")

    # 3. Uniqueness Check
    if len(set(password)) <= 3 and len(password) > 3:
        score -= 2
        feedback.append(
            "• Password lacks uniqueness (too many repeating characters).")

    common_sequences = ["1234", "qwerty", "password", "admin"]
    if any(seq in password.lower() for seq in common_sequences):
        score -= 2
        feedback.append("• Avoid common words or predictable sequences.")

    # Strength Assessment
    if score >= 5:
        strength = "STRONG 💪"
    elif 3 <= score < 5:
        strength = "MEDIUM ⚠️"
    else:
        strength = "WEAK ❌"

    return strength, feedback


def generate_strong_alternative(length=14):
    alphabet = string.ascii_letters + string.digits + "!@#$%^*_+-="
    while True:
        password = ''.join(secrets.choice(alphabet) for _ in range(length))
        if (any(c.islower() for c in password)
                and any(c.isupper() for c in password)
                and any(c.isdigit() for c in password)
                and any(c in "!@#$%^*_+-=" for c in password)):
            return password

# --- Interactive Terminal UI ---


def main():
    init_db()  # Setup database on execution

    print("==========================================")
    print("🔒 ADVANCED PASSWORD ANALYZER WITH DATABASE")
    print("==========================================\n")

    user_id = input("Enter your Username / User ID: ").strip().lower()
    if not user_id:
        print("User ID cannot be blank.")
        return

    user_pass = input("Enter a password to evaluate: ").strip()
    if not user_pass:
        print("Password cannot be blank.")
        return

    # --- Database Reuse Validation Step ---
    if is_password_reused(user_id, user_pass):
        print("\n🛑 SECURITY ALERT: You cannot reuse this password!")
        print("This password matches a previously used password in our database records.")
        print(
            f"💡 Try a fresh alternative like: {generate_strong_alternative()}")
        print("==========================================")
        return

    # --- Structural Analysis Step ---
    strength, suggestions = analyze_password(user_pass)

    print("\n--- Evaluation Result ---")
    print(f"Overall Strength: {strength}")

    if suggestions:
        print("\nSuggestions for Improvement:")
        for line in suggestions:
            print(line)
        print(f"\n💡 Recommended Alternative: {generate_strong_alternative()}")
    else:
        print("\n✨ Excellent! Your password meets all basic complexity standards.")
        # Save the secure hash if the password is good and unique
        save_password_hash(user_id, user_pass)
        print("💾 Secure password hash recorded in history database successfully.")

    print("==========================================")


if __name__ == "__main__":
    main()
